use std::fs;

fn main() {
    let content = fs::read_to_string("./input.txt").expect("file not read");
    let mut games: Vec<Vec<(u32, &str)>> = Vec::new();
    let mut c_red = 0;
    let mut c_green = 0;
    let mut c_blue = 0;

    for line in content.lines() {
        let parts: Vec<&str> = line.split(':').collect();
        println!("{}", parts[1]);

        let mut current_game: Vec<(u32, &str)> = Vec::new();
        for item in parts[1].split([',', ';']) {
            let words: Vec<&str> = item.split_whitespace().collect();
            if words.is_empty() {
                continue;
            }

            let count: u32 = words[0].parse().unwrap();
            let color: &str = words[1];
            current_game.push((count, color));
        }
        games.push(current_game);
    }

    for game in &games {
        for (count, color) in game {
            match *color {
                "red" => c_red += *count,
                "green" => c_green += *count,
                "blue" => c_blue += *count,
                _ => {}
            }
        }
    }

    println!("{} {} {}", c_red, c_green, c_blue);
}
